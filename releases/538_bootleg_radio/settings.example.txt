# Bootleg Radio (538) - example settings.txt
#
# Copy this file to the ROOT of your USB stick and rename it to "settings.txt".
# It is read once each time the stick is mounted, after the filesystem mounts.
# Every line is optional; delete or comment out anything you do not want to
# change, and the card keeps its default. Lines starting with # or ; are
# comments. Keys are case-insensitive. The values below are all the defaults, so
# this file as-is changes nothing.

# Power-on "tune" mode: loop | sync
#   loop  a new station starts at its loop start point
#   sync  a new station keeps the proportional position (synchronised jump)
# Tapping B4 on the 8mu changes the running mode; this is the power-on value.
mode = sync

# What happens at the end of a station: loop | one-shot | ping-pong
#   loop       wrap back to the loop start (default)
#   one-shot   play once and stop until retriggered
#   ping-pong  alternate forward and reverse passes
loopMode = loop

# F7 performance mode: off | scan | repeat
#   off     normal playback
#   scan    advance station on the clock
#   repeat  retrigger the current loop on the clock
performanceMode = off

# Clock pulses per scan/repeat step, 1-16
perfDiv = 1

# Bank to start on, 0-31 (0 = the root directory; 1.. = subfolders in order)
bank = 0

# Speed control range: 2oct | 1oct | fine | off
#   2oct  -2..+2 octaves (default)
#   1oct  -1..+1 octave
#   fine  about -1..+1 semitone
#   off   fixed 1x; X, CV in 2 and 8mu F1 do nothing
# The 8mu's F2 overrides this while it is above its dead zone.
speedRange = 2oct

# Internal clock tempo in BPM, 30-240 (used when Pulse in 1 is unpatched)
clockBpm = 120

# Clock pulses per triangle cycle on CV out 2, 1-256
#   40 = one cycle every 20 seconds at 120 BPM
triPulses = 40

# Pulse out 1: gate | tick | clock | off
#   gate   high while the loop plays, low during the wrap/seek gap
#   tick   a short pulse at each loop wrap
#   clock  the active clock pulse
#   off    always low
pulse1 = gate

# Pulse out 2: gate | tick | clock | off
pulse2 = tick

# LED display: verbose | quiet
#   verbose  level, playing, station-change, mounted, underrun/fault
#   quiet    level, playing and mounted only (LEDs 3 and 5 stay dark)
led = verbose

# Pulse in 2 / TRS MIDI remote: auto | on | off
#   auto  retrigger until a Music Thing 8mu is heard, then MIDI
#   on    always the 8mu input (Pulse in 2 no longer retriggers)
#   off   always the external retrigger (MIDI ignored)
remote = auto

# Tuning static: off | on
#   adds a short filtered noise burst when you change station or bank
# Holding B2/B3 on the 8mu changes the running state; this is the power-on value.
static = off

# Static level, 0-100
staticLevel = 50
