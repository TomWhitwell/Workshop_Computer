cmake_minimum_required(VERSION 3.13)
include(${CMAKE_CURRENT_LIST_DIR}/pico_sdk_import.cmake)
project(spatial_disorientation C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
set(SPATIAL_HRTF_TAPS "64" CACHE STRING "Beta HRTF filter length: 64")
if(NOT SPATIAL_HRTF_TAPS STREQUAL "64")
    message(FATAL_ERROR "Beta requires 64 taps")
endif()
option(SPATIAL_ELEVATION_SOLO "Diagnostic source-A-only build (not the beta distribution)" OFF)
add_executable(spatial_disorientation src/main.cpp src/usb_descriptors.c)
target_include_directories(spatial_disorientation PRIVATE ${CMAKE_CURRENT_LIST_DIR}/vendor/ComputerCard ${CMAKE_CURRENT_LIST_DIR}/vendor/EightMU src)
target_compile_options(spatial_disorientation PRIVATE -fstack-usage -Wdouble-promotion -Wfloat-conversion -Wall -Wextra)
target_compile_definitions(spatial_disorientation PRIVATE PICO_XOSC_STARTUP_DELAY_MULTIPLIER=64 SPATIAL_HRTF_TAPS=${SPATIAL_HRTF_TAPS} SPATIAL_ELEVATION_SOLO=$<BOOL:${SPATIAL_ELEVATION_SOLO}>)
target_link_options(spatial_disorientation PRIVATE -Wl,--print-memory-usage)
target_link_libraries(spatial_disorientation pico_unique_id pico_stdlib hardware_dma hardware_i2c hardware_pwm hardware_adc hardware_spi hardware_flash pico_multicore tinyusb_host tinyusb_device tinyusb_board)
pico_enable_stdio_usb(spatial_disorientation 0)
pico_enable_stdio_uart(spatial_disorientation 0)
pico_set_binary_type(spatial_disorientation copy_to_ram)
pico_add_extra_outputs(spatial_disorientation)
