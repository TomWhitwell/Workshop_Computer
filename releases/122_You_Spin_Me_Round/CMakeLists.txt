cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(you_spin_me_round C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(you_spin_me_round src/main.cpp)
target_include_directories(you_spin_me_round PRIVATE vendor/ComputerCard)
target_compile_options(you_spin_me_round PRIVATE -Wdouble-promotion -Wfloat-conversion -Wall -Wextra)
target_compile_definitions(you_spin_me_round PRIVATE PICO_XOSC_STARTUP_DELAY_MULTIPLIER=64)
target_link_options(you_spin_me_round PRIVATE -Wl,--print-memory-usage)
target_link_libraries(you_spin_me_round pico_unique_id pico_stdlib hardware_dma hardware_i2c hardware_pwm hardware_adc hardware_spi)
pico_enable_stdio_usb(you_spin_me_round 0)
pico_enable_stdio_uart(you_spin_me_round 0)
pico_set_binary_type(you_spin_me_round copy_to_ram)
pico_add_extra_outputs(you_spin_me_round)
