cmake_minimum_required(VERSION 3.24)
option(PEDAL_BROWSER_BASE "Prepare reusable browser firmware objects" OFF)
option(PEDAL_HOST "Build host validation and renderer" OFF)
option(PEDAL_DIAGNOSTIC "Build a direct stereo-through hardware diagnostic" OFF)
option(PEDAL_TELEMETRY "Enable USB CDC telemetry" OFF)
if(NOT PEDAL_HOST)
  if(NOT DEFINED PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
  endif()
  if(NOT PICO_SDK_PATH)
    message(FATAL_ERROR "Set PICO_SDK_PATH to Pico SDK 2.1.1 or newer")
  endif()
  include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(pedal_lab C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(PEDAL_PROGRAMS
  "${CMAKE_CURRENT_SOURCE_DIR}/programs/00_55_gallon_saint.spn;${CMAKE_CURRENT_SOURCE_DIR}/programs/01_last_known_copy.spn;${CMAKE_CURRENT_SOURCE_DIR}/programs/02_ghost_spring.spn;${CMAKE_CURRENT_SOURCE_DIR}/programs/03_gravity_clerk.spn;${CMAKE_CURRENT_SOURCE_DIR}/programs/04_cold_case.spn;${CMAKE_CURRENT_SOURCE_DIR}/programs/06_reverse_witness.spn"
  CACHE STRING "Semicolon-separated bank of 1 to 6 SpinASM files")
set(BANK_DIR ${CMAKE_CURRENT_BINARY_DIR}/bank)
if(PEDAL_BROWSER_BASE)
  file(MAKE_DIRECTORY ${BANK_DIR})
  configure_file(src/browser_bank.hpp ${BANK_DIR}/bank.hpp COPYONLY)
  file(WRITE ${BANK_DIR}/bank.json "[]")
  add_custom_target(effect_bank)
else()
add_custom_command(OUTPUT ${BANK_DIR}/bank.hpp ${BANK_DIR}/bank.json
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/compile_bank.py ${PEDAL_PROGRAMS} -o ${BANK_DIR}
  DEPENDS ${PEDAL_PROGRAMS} tools/compile_bank.py vendor/assembler/assembler.py algorithms/catalogue.json
  VERBATIM)
add_custom_target(effect_bank DEPENDS ${BANK_DIR}/bank.hpp ${BANK_DIR}/bank.json)
endif()
add_library(effect_headers INTERFACE)
target_include_directories(effect_headers INTERFACE ${BANK_DIR} src generated vendor/dsp-reference/include)
add_dependencies(effect_headers effect_bank)
if(PEDAL_HOST)
  enable_testing()
  add_test(NAME compiler COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_compiler.py)
  add_test(NAME catalogue COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_catalogue.py)
  add_library(reference_engine STATIC vendor/dsp-reference/engine.cpp vendor/dsp-reference/reference_model.cpp)
  target_include_directories(reference_engine PUBLIC vendor/dsp-reference/include)
  add_executable(validate tests/validate.cpp)
  target_link_libraries(validate effect_headers reference_engine)
  add_test(NAME differential COMMAND validate ${BANK_DIR})
  add_executable(controls tests/controls.cpp)
  target_include_directories(controls PRIVATE tests/mock)
  target_link_libraries(controls effect_headers)
  add_test(NAME controls COMMAND controls)
  add_executable(worker tests/worker.cpp)
  find_package(Threads REQUIRED)
  target_link_libraries(worker effect_headers Threads::Threads)
  add_test(NAME worker COMMAND worker)
  add_executable(audio_checks tests/audio.cpp)
  target_link_libraries(audio_checks effect_headers)
  add_test(NAME audio COMMAND audio_checks)
  add_executable(arm_vectors tests/arm_vectors.cpp)
  target_link_libraries(arm_vectors effect_headers)
  add_executable(audio_vectors tests/audio_vectors.cpp)
  target_link_libraries(audio_vectors effect_headers)
  find_program(PEDAL_NODE node)
  if(PEDAL_NODE)
    add_test(NAME browser_preview COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/browser_preview.py $<TARGET_FILE:audio_vectors> ${BANK_DIR}/bank.json ${PEDAL_NODE})
  endif()
  add_executable(virtual_vectors tests/virtual_vectors.cpp)
  target_link_libraries(virtual_vectors effect_headers)
  add_executable(render tools/render.cpp)
  target_link_libraries(render effect_headers)
else()
  pico_sdk_init()
  if(PEDAL_DIAGNOSTIC)
    add_executable(pedal_lab src/diagnostic.cpp)
  else()
    add_executable(pedal_lab src/main.cpp)
  endif()
  if(PEDAL_BROWSER_BASE)
    target_sources(pedal_lab PRIVATE src/browser_bank_stub.cpp)
  endif()
  target_link_libraries(pedal_lab effect_headers pico_stdlib pico_multicore pico_unique_id hardware_dma hardware_i2c hardware_pwm hardware_adc hardware_spi)
  target_include_directories(pedal_lab PRIVATE vendor/ComputerCard)
  target_compile_options(pedal_lab PRIVATE -O3 -Wall -Wextra -Wdouble-promotion -Wfloat-conversion)
  target_compile_definitions(pedal_lab PRIVATE PICO_XOSC_STARTUP_DELAY_MULTIPLIER=64)
  target_link_options(pedal_lab PRIVATE -Wl,--print-memory-usage)
  if(NOT PEDAL_DIAGNOSTIC)
    target_link_options(pedal_lab PRIVATE -Wl,--undefined=spin_resample_frame)
  endif()
  # Keep audio code and tables out of XIP. RAM use includes the 64KiB delay RAM.
  pico_set_binary_type(pedal_lab copy_to_ram)
  if(PEDAL_TELEMETRY OR PEDAL_DIAGNOSTIC)
    pico_enable_stdio_usb(pedal_lab 1)
    target_compile_definitions(pedal_lab PRIVATE PEDAL_TELEMETRY=1 COMPUTERCARD_MAIN_LOOP_CALLBACK=1)
  else()
    pico_enable_stdio_usb(pedal_lab 0)
  endif()
  pico_enable_stdio_uart(pedal_lab 0)
  pico_add_extra_outputs(pedal_lab)
endif()
