cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(Departures C CXX ASM)
configure_file(NOTICE.txt NOTICE.txt COPYONLY)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(Departures src/main.cpp)
target_include_directories(Departures PRIVATE src vendor/ComputerCard)
target_compile_options(Departures PRIVATE -O3 -Wall -Wextra -Wdouble-promotion -Wfloat-conversion)
target_compile_definitions(Departures PRIVATE PICO_XOSC_STARTUP_DELAY_MULTIPLIER=64)
target_link_options(Departures PRIVATE -Wl,--print-memory-usage)
target_link_libraries(Departures pico_unique_id pico_stdlib pico_rand hardware_dma hardware_i2c
    hardware_pwm hardware_adc hardware_spi hardware_clocks hardware_vreg)
pico_enable_stdio_usb(Departures 0)
pico_enable_stdio_uart(Departures 0)
pico_set_binary_type(Departures copy_to_ram)
pico_add_extra_outputs(Departures)

option(DEPARTURES_BENCH "Export hardware field offsets for the ARM timing harness" OFF)
if(DEPARTURES_BENCH)
    target_sources(Departures PRIVATE test/bench_layout.cpp)
    target_link_options(Departures PRIVATE -Wl,--undefined=bench_offsets)
endif()
